function calculateSalary(basicSalary: number) {

    let hra = basicSalary * 0.20;
    let da = basicSalary * 0.10;
    let totalSalary = basicSalary + hra + da;

    console.log("Basic Salary: " + basicSalary);
    console.log("HRA: " + hra);
    console.log("DA: " + da);
    console.log("Total Salary: " + totalSalary);
}
